/* GHOST PROTOCOL - RESILIENCE TEST MODULE 
   Target: AIC Network (7 Nodes)
   Objective: Validate BFT (Byzantine Fault Tolerance) & Veto Latency
*/

const ResilienceTest = {
    originalNodes: 7,
    
    // 1. Simula il distacco fisico di nodi casuali
    async triggerNodeBlackout(count = 2) {
        addLog('NETWORK', `INITIATING BLACKOUT TEST: Dropping ${count} nodes...`, 'security');
        
        let activeNodes = this.originalNodes - count;
        document.getElementById('nodes').textContent = `${activeNodes}/7 ACTIVE`;
        document.getElementById('nodes-bar').style.width = (activeNodes / 7 * 100) + '%';
        
        // Verifica latenza di rinegoziazione
        const startTime = performance.now();
        await this.verifyConsensus(activeNodes);
        const endTime = performance.now();
        
        vetoLatency = (endTime - startTime).toFixed(2);
        document.getElementById('latency').textContent = `${vetoLatency} ms`;
        
        addLog('RESILIENCE', `CONSENSUS RE-ESTABLISHED WITH ${activeNodes} NODES`, 'ghost');
    },

    // 2. Stress Test H-VAR: Iniezione di dati non conformi
    injectAnomalousData() {
        addLog('VALIDATION', 'INJECTING NON-NSR COMPLIANT DATA...', 'critical');
        
        // La divergenza deve reagire
        let drift = 0.0850; 
        document.getElementById('divergence').textContent = drift.toFixed(4);
        document.getElementById('divergence-bar').style.width = '40%';
        
        // Il QEK deve attivare il Veto se H-VAR supera la soglia
        if (drift > hvar) {
            addLog('GHOST', 'H-VAR THRESHOLD EXCEEDED: Veto triggered', 'highlight');
            this.stabilizeSystem();
        }
    },

    // 3. Funzione di auto-ripristino (Healing)
    stabilizeSystem() {
        setTimeout(() => {
            document.getElementById('divergence').textContent = "0.0000";
            document.getElementById('divergence-bar').style.width = '100%';
            addLog('SYSTEM', 'SYSTEM STABILIZED: Divergence corrected by QEK', 'optimal');
        }, 3000);
    },

    verifyConsensus: (n) => new Promise(resolve => setTimeout(resolve, Math.random() * 3))
};
